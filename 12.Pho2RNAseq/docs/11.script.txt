#### 0.3.Trim-20240123

Raw reads were filtered and trimmed to get high quality reads using Trimmomatic (v0.39). Briefly, the bases with quality less than 20 at the start or end of a read were cut off, and reads with length shorter than 60bp were dropped.

Since there are so many samples, I used a `for loop` to do trimmomatic.

```
#!/bin/bash

# we are using trimmomatic.jar from /usr/local/software/trimmomatic/0.39/trimmomatic-0.39.jar

input_dir="/rds/user/cx264/hpc-work/project/2.Jeongmin/01.Saskia/1.data/X204SC23091440-Z01-F002_02/01.RawData/"
output_dir="/rds/user/cx264/hpc-work/project/2.Jeongmin/01.Saskia/1.data/X204SC23091440-Z01-F002_02/02.CleanData/"

# run trimmomatic iterate through each sample
for fq1 in "$input_dir"/*/*_1.fq.gz; do
    fq2="${fq1/_1.fq.gz/_2.fq.gz}" # get the corresponding paired-end file

    # extract sample name
    sample_name=$(basename "$fq1" | cut -d'_' -f1-2)

    # set output files
    output_fwd_paired="${sample_name}_1P.fq.gz"
    output_fwd_unpaired="${sample_name}_1U.fq.gz"
    output_rev_paired="${sample_name}_2P.fq.gz"
    output_rev_unpaired="${sample_name}_2U.fq.gz"

    # Run Trimmomatic
    echo "processing ${sample_name}"
    cd $output_dir
    mkdir ${sample_name} && cd ${sample_name}
    java -jar /usr/local/software/trimmomatic/0.39/trimmomatic-0.39.jar PE -threads 4 -summary ${sample_name}.summary "$fq1" "$fq2" \
        "$output_fwd_paired" "$output_fwd_unpaired" "$output_rev_paired" "$output_rev_unpaired" \
        LEADING:20 TRAILING:20 SLIDINGWINDOW:4:20 MINLEN:60
    echo "${sample_name} finished"
done

echo "Trimmomatic process completed."
```

An in-house python script was used to summarize mapping results.`for i in {1..32}; do echo r2310_$i; python quality.py r2310_$i/Overall_mapping_summary.log; done`. (input was the log from mapinsights)

```
import re
import sys

def extract_info(file_path):
    with open(file_path, 'r') as file:
        content = file.read()

    # Extracting relevant information using regular expressions
    total_reads = int(re.search(r'Total number of reads = (\d+)', content).group(1))
    mapped_reads = int(re.search(r'Total number of mapped reads = (\d+)', content).group(1))
    supplementary_alignments = int(re.search(r'Supplementary-alignments = (\d+)', content).group(1))
    mismatch_rate = float(re.search(r'Mismatch rate = (\d+\.\d+)', content).group(1))
    mean_depth_of_coverage = float(re.search(r'Mean depth of coverages = (\d+\.\d+)', content).group(1))
    mean_mapping_quality = float(re.search(r'Mean mapping quality = (\d+\.\d+)', content).group(1))

    return total_reads, mapped_reads, supplementary_alignments, mismatch_rate, mean_depth_of_coverage, mean_mapping_quality

def main():
    if len(sys.argv) != 2:
        print("Usage: python mypython.py input.txt")
        sys.exit(1)

    input_file_path = sys.argv[1]

    results = extract_info(input_file_path)

    # Corrected ratio calculation for Python 2
    ratio_mapped_reads = float(results[1] - results[2]) / (results[0] - results[2])

    print("\t".join(map(str, [
        results[1] - results[2],
        ratio_mapped_reads,
        results[3],
        results[4],
        results[5]
    ])))

if __name__ == "__main__":
    main()
```

#cat paste.py
import csv
import sys

def combine_files(file1_path, file2_path, output_path):
    # Read the second file and create a dictionary with the first column as the key and the second column as the value.
    file2_data = {}
    with open(file2_path, 'r') as file2:
        reader = csv.reader(file2, delimiter='\t')
        for row in reader:
            if row:  # Ensure the row is not empty
                file2_data[row[0]] = row[1]

    # Open the first file to read and a new file to write the combined data.
    with open(file1_path, 'r') as file1, open(output_path, 'w', newline='') as output:
        reader = csv.reader(file1, delimiter='\t')
        writer = csv.writer(output, delimiter='\t')

        for row in reader:
            if row:  # Ensure the row is not empty
                # Check if the first column of the first file exists in the second file's dictionary.
                key = row[0]
                if key in file2_data:
                    # Append the second column from the second file to the current row.
                    row.append(file2_data[key])
                # Write the updated row to the output file.
                writer.writerow(row)

    print(f"The second column from {file2_path} has been successfully appended to {file1_path} where the first columns match.")

if __name__ == '__main__':
    if len(sys.argv) != 4:
        print("Usage: python script.py first.txt second.txt output.txt")
    else:
        file1_path = sys.argv[1]
        file2_path = sys.argv[2]
        output_path = sys.argv[3]
        combine_files(file1_path, file2_path, output_path)
